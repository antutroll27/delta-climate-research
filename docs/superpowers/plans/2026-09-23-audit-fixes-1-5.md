## Ground rules for whoever executes this

- **Never push, open a PR, or merge without the owner's explicit word.** `git push origin HEAD:main`
  carries every commit beneath HEAD; that is how unfinished work reached production on 2026-09-06.
- **Never switch the main checkout's branch** (`/Volumes/VSTSAMPLES/Projects/Angad`). Another Claude
  session was editing files there on 2026-09-23. PR 1 and PR 2 are built in `~/delta-worktrees/audit-fixes`.
  Before touching the main checkout at all, run `git -C /Volumes/VSTSAMPLES/Projects/Angad status --porcelain --untracked-files=no`
  and treat any modification you did not make as someone else's live work.
- **Stage only named paths** (`git add -- <path>`), never `git add -A`. Follow-ups are new commits, never `--amend`.
- **Commit messages via heredoc**, ending with the trailer
  `Co-Authored-By: Claude Opus 5.5 (1M context) <noreply@anthropic.com>`.
- **Every `.py` passes strict mypy:** `python3 -m mypy` must print `Success` before any Python commit.
- **Playwright runs in the FOREGROUND** with a Bash timeout of at most 540000 ms, two runs maximum per
  step. Never background a Playwright run and wait for a notification (that stalled for hours, twice).
- **Full `npm run verify` is a controller job** (10–25 min): run it as a background task and wait for
  its completion notice. Subagents run only the targeted commands in their task.
- **Never `git stash`**: the stash stack is shared across worktrees. Never `git reset --hard` on the
  exFAT volume.
- **Controller-only tasks** are marked. They involve confidential names, shared-checkout operations,
  audits, or pushes, and must not be delegated.

```js
import assert from 'node:assert/strict';
import test from 'node:test';
import { readFileSync, readdirSync } from 'node:fs';

/*
 * 2026-09-23 audit, item 1. The ward route rendered <Base> without `nativeScroll`,
 * so Lenis started on the OBOS page and cancelled every wheel event inside its side
 * panels — on a page whose own body never scrolls (HeatMapStage sets

 */
const PAGES = 'src/pages/heat-map';
const astroFiles = readdirSync(PAGES, { recursive: true })
  .filter((f) => String(f).endsWith('.astro'))
  .map((f) => join(PAGES, String(f)));

test('every page that renders the OBOS stage opts out of Lenis', () => {
  const stages = astroFiles.filter((f) => readFileSync(f, 'utf8').includes('<HeatMapStage'));
  assert.ok(stages.length >= 1, `no page under ${PAGES} renders <HeatMapStage> — did the route move?`);
  for (const file of stages) {
    const base = readFileSync(file, 'utf8').match(/<Base\b[^>]*>/s);
    assert.ok(base, `${file}: no <Base> element`);
    assert.match(base[0], /\bnativeScroll\b/,
      `${file}: <Base> lacks nativeScroll, so Lenis will swallow wheel events in the panels`);

});
```

```ts
 * 2026-09-23 audit, item 1. The ward route started Lenis, whose window-level wheel
 * listener calls preventDefault() on every vertical wheel event, so no side panel

 * where Lenis never starts. Map zoom still worked — MapLibre handles its own wheel
 * first — which is why nobody noticed.

 * exactly the event Lenis intercepts. That is how this shipped green.
 *
 * WHY THE LAYERS PANE AT 1280x480. It is light — no solar computation, which is
 * CPU-marginal on CI's two cores — and at this height its tree overflows (measured:
 * 353 px of content in 297 px at 1280x520; a shorter viewport overflows more).

  await page.setViewportSize({ width: 1280, height: 480 });
  await page.goto('/heat-map/in/kolkata/ballygunge/', { waitUntil: 'domcontentloaded' });
  await expect(page.locator('.maplibregl-map')).toBeVisible({ timeout: 30_000 });

  const size = await tree.evaluate((el) => ({ scroll: el.scrollHeight, client: el.clientHeight }));

  for (let i = 0; i < 4; i++) {

    await page.waitForTimeout(100);
  }
  await expect.poll(() => tree.evaluate((el) => el.scrollTop), { timeout: 5_000 }).toBeGreaterThan(0);
});
```

```python
def power_stamp(iso_utc: str, lon: float) -> str:
    """The POWER `YYYYMMDDHH` stamp nearest `iso_utc` in local solar time at `lon`.

    `iso_utc` is an ISO-8601 instant in UTC, with or without a trailing `Z`, e.g.
    `2026-04-12T23:28:34`. Local solar time is UTC + lon/15 hours; the nearest hour
    is taken by adding 30 minutes and truncating, so 23:40 local reads the next
    day's 00 stamp — the date rolls with the hour, never separately.
    """
    t = dt.datetime.fromisoformat(iso_utc.rstrip("Z"))
    lst = t + dt.timedelta(hours=lon / 15.0)
    return (lst + dt.timedelta(minutes=30)).strftime("%Y%m%d%H")
```

```python
#!/usr/bin/env python3
"""P1-P5 of spec section 2.4: did the forcing-date correction touch only what it should?

Run from the PR 2 worktree root. Compares the pre-change snapshot with the rebuilt
artefacts, prints a markdown report for the PR description, and exits 1 if P1, P2 or
P3 fails. P4 and P5 are reported, never fatal: their decision rule is a human one
(confirm the fix, then publish the honest number).
"""
from __future__ import annotations

import csv
import datetime as dt
import json
import sys
from pathlib import Path
from typing import Any

SNAP = Path.home() / ".cache/delta-climate/audits/2026-09-23-forcing/before"
CAL = Path("data/calibration")
ORIGINAL = ["date", "phase", "local_solar_hour", "view_delta", "usable_frac", "urban_mean",
            "rural_strict", "suhii", "tAir", "rh", "wind", "cloud"]
FORCING = {"tAir", "rh", "wind", "cloud"}
SURFACE = ["lst_mean_c", "lst_sd_c", "cells", "cell_frac", "fvc", "albedo", "built"]


def met(p: Path) -> dict[tuple[str, str], dict[str, str]]:
    with open(p, newline="") as fh:
        return {(r["date"], r["phase"]): r for r in csv.DictReader(fh)}


def old_key(date: str, lsh: str) -> str:
    """The stamp the pre-fix code built: UTC date + rounded local solar hour."""
    h, d = int(round(float(lsh))), dt.date.fromisoformat(date)
    if h >= 24:
        h, d = h - 24, d + dt.timedelta(days=1)
    return d.strftime("%Y%m%d") + f"{h:02d}"


def leaves(o: Any, path: str = "") -> dict[str, Any]:
    out: dict[str, Any] = {}
    if isinstance(o, dict):
        for k, v in o.items():
            out.update(leaves(v, f"{path}.{k}" if path else str(k)))
    elif isinstance(o, list):
        for i, v in enumerate(o):
            out.update(leaves(v, f"{path}[{i}]"))
    else:
        out[path] = o
    return out


fatal: list[str] = []
L = ["## Forcing-date correction: P1-P5 (spec section 2.4)", ""]

old, new = met(SNAP / "met-forcing.csv"), met(CAL / "met-forcing.csv")
moved = {k for k, r in new.items() if r["power_stamp"] != old_key(k[0], r["local_solar_hour"])}
changed = {k for k in old if k in new and any(old[k][c] != new[k][c] for c in ORIGINAL)}
touched = sorted({c for k in changed for c in ORIGINAL if old[k][c] != new[k][c]})
p1 = (set(old) == set(new) and len(moved) == 21 and all(k[1] == "night" for k in moved)
      and changed <= moved and set(touched) <= FORCING)
L += [f"- **P1** rows whose POWER stamp moved: {len(moved)} (pre-registered: 21, all night)",
      f"- **P1** rows changed in the original twelve columns: {len(changed)}; columns: {', '.join(touched) or 'none'}",
      f"- **P1** {'holds' if p1 else 'VIOLATED'}"]
if not p1:
    fatal.append("P1")

# Compared BY POSITION: (date, phase, ward) is not unique in this file (82 repeats,
# measured 2026-09-23), so a dict keyed on it would silently drop rows. The build is
# deterministic and its row order follows met-forcing.csv, whose order does not change.
wo_a, wo_b = json.load(open(SNAP / "ward-observations.json")), json.load(open(CAL / "ward-observations.json"))
ra, rb = wo_a["rows"], wo_b["rows"]
IDENT = ("date", "phase", "ward", "sensor", "hour")
header_same = {k: v for k, v in wo_a.items() if k != "rows"} == {k: v for k, v in wo_b.items() if k != "rows"}
aligned = len(ra) == len(rb) and all(all(x[c] == y[c] for c in IDENT) for x, y in zip(ra, rb))
wo_changed = [(x, y) for x, y in zip(ra, rb) if x != y]
wo_fields = sorted({f for x, y in wo_changed for f in set(x) | set(y) if x.get(f) != y.get(f)})
surface_moved = sum(1 for x, y in zip(ra, rb) if any(x.get(c) != y.get(c) for c in SURFACE))
p2 = (header_same and aligned and surface_moved == 0 and set(wo_fields) <= FORCING
      and all((x["date"], x["phase"]) in moved for x, _ in wo_changed))
L += [f"- **P2** rows aligned: {aligned}; header unchanged: {header_same}; rows changed: {len(wo_changed)}; "
      f"fields: {', '.join(wo_fields) or 'none'}; surface fields moved in {surface_moved} rows",
      f"- **P2** {'holds' if p2 else 'VIOLATED'}"]
if not p2:
    fatal.append("P2")

la = leaves(json.load(open(SNAP / "model-accuracy.json")))
lb = leaves(json.load(open(CAL / "model-accuracy.json")))
diff_paths = sorted(p for p in set(la) | set(lb) if la.get(p) != lb.get(p))
forbidden = [p for p in diff_paths if any(s in p.lower() for s in ("peak", "morning", "landsat"))]
p3 = not forbidden
L += [f"- **P3** changed paths in model-accuracy.json: {len(diff_paths)}",
      *[f"  - `{p}`: {la.get(p)} -> {lb.get(p)}" for p in diff_paths],
      f"- **P3** {'holds' if p3 else 'VIOLATED: ' + ', '.join(forbidden)}"]
if not p3:
    fatal.append("P3")

n0 = json.load(open(SNAP / "model-accuracy.json"))["ward_scale"]["strata"]["night"]
n1 = json.load(open(CAL / "model-accuracy.json"))["ward_scale"]["strata"]["night"]
L.append(f"- **P4** night bias {n0['bias_K']} -> {n1['bias_K']} K: "
         f"{'moved up, as predicted' if n1['bias_K'] > n0['bias_K'] else 'DID NOT move up'}")
for key, label in (("rmse_K", "RMSE"), ("loo_overpass_rmse_K", "leave-one-overpass-out RMSE"),
                   ("ceiling_rmse_K", "ceiling"), ("n_scenes", "n_scenes")):
    L.append(f"- **P5** night {label}: {n0[key]} -> {n1[key]}")
p5 = (n1["rmse_K"] <= n0["rmse_K"] + 0.02 and n1["loo_overpass_rmse_K"] <= n0["loo_overpass_rmse_K"] + 0.02)
L.append(f"- **P5** {'holds (flat or better, within 0.02 K)' if p5 else 'FAILED: confirm the fix, then publish the honest number'}")

print("\n".join(L))
sys.exit(1 if fatal else 0)
```

```python
def verify_provenance() -> int:
    """Does INPUT_COMMITS still name the commit that last changed each input?

    Needs full history, so it refuses a shallow clone rather than report every input as
    last changed by the tip. Local only: CI never runs it. INPUT_COMMITS is edited by
    hand when this reports a mismatch; the script never rewrites its own source.
    """
    if git("rev-parse", "--is-shallow-repository").strip() == "true":
        print("  refused: this is a shallow clone, where every file's last commit is the tip")
        return 1
    wrong = []
    for path, recorded in INPUT_COMMITS.items():
        rel = os.path.relpath(path, ROOT)
        last = git("log", "-1", "--format=%H", "--", rel).strip()[:7]
        if last != recorded:
            wrong.append(f"{rel}: INPUT_COMMITS says {recorded}, git history says {last}")
    for line in wrong:
        print(f"  MISMATCH {line}")
    if wrong:
        return 1
    print(f"  provenance ok: {len(INPUT_COMMITS)} recorded commits match git history")
    return 0
```
